cmake_minimum_required(VERSION 3.10)
project(rm27_schurvins_headless LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_BUILD_TYPE Release)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT IS_DIRECTORY "${SCHUR_SOURCE}/SchurVINS" OR NOT IS_DIRECTORY "${SCHUR_DEP_LIB}")
  message(FATAL_ERROR "SCHUR_SOURCE must contain pinned sources and SCHUR_DEP_LIB target opengv libraries")
endif()
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs video calib3d highgui)
find_package(Eigen3 REQUIRED)
find_package(Boost REQUIRED COMPONENTS filesystem system)
find_package(Threads REQUIRED)
find_library(YAML_CPP_LIBRARY yaml-cpp)
if(NOT YAML_CPP_LIBRARY)
  message(FATAL_ERROR "Target yaml-cpp library is required")
endif()
if(CMAKE_CROSSCOMPILING)
  set(TARGET_OPENCV_INCLUDES)
  foreach(path ${OpenCV_INCLUDE_DIRS})
    if(IS_ABSOLUTE "${path}" AND EXISTS "${CMAKE_SYSROOT}${path}")
      list(APPEND TARGET_OPENCV_INCLUDES "${CMAKE_SYSROOT}${path}")
    else()
      list(APPEND TARGET_OPENCV_INCLUDES "${path}")
    endif()
  endforeach()
  set(OpenCV_INCLUDE_DIRS ${TARGET_OPENCV_INCLUDES})
endif()
set(SVO "${SCHUR_SOURCE}/SchurVINS")
include_directories(${EIGEN3_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS}
  "${SCHUR_SOURCE}/minkindr/minkindr/include" "${SCHUR_SOURCE}/eigen_checks/include"
  "${SCHUR_SOURCE}/opengv/include" "${SCHUR_SOURCE}/fast_neon/include")
foreach(component svo svo_common svo_vio_common svo_direct svo_img_align svo_tracker rpg_common)
  include_directories("${SVO}/${component}/include")
endforeach()
foreach(component vikit_common vikit_cameras vikit_solver)
  include_directories("${SVO}/vikit/${component}/include")
endforeach()
add_definitions(-DSVO_USE_OPENGV -DSVO_DEPTHFILTER_IN_REPROJECTOR)
add_compile_options(-O3 -pthread -fomit-frame-pointer -fno-signed-zeros -fno-math-errno -funroll-loops -ffast-math -fno-finite-math-only -Wno-deprecated-declarations)
string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" ARCH)
set(FAST_SOURCES "${SCHUR_SOURCE}/fast_neon/src/fast_9.cpp" "${SCHUR_SOURCE}/fast_neon/src/fast_10.cpp" "${SCHUR_SOURCE}/fast_neon/src/fast_10_score.cpp" "${SCHUR_SOURCE}/fast_neon/src/nonmax_3x3.cpp")
if(ARCH MATCHES "^(aarch64|arm64)$")
  add_compile_options(-march=armv8-a+simd)
  add_definitions(-DHAVE_FAST_NEON)
  list(APPEND FAST_SOURCES "${SCHUR_SOURCE}/fast_neon/src/faster_corner_9_neon.cpp")
elseif(ARCH MATCHES "^(x86_64|amd64)$")
  add_compile_options(-mmmx -msse -msse2 -msse3 -mssse3 -mno-avx)
  list(APPEND FAST_SOURCES "${SCHUR_SOURCE}/fast_neon/src/faster_corner_10_sse.cpp")
else()
  message(FATAL_ERROR "Unsupported target: ${CMAKE_SYSTEM_PROCESSOR}")
endif()
set(CORE_SOURCES ${FAST_SOURCES} "${SVO}/rpg_common/src/fs.cpp" "${SVO}/vikit/vikit_solver/src/robust_cost.cpp")
foreach(component svo svo_common svo_direct svo_tracker)
  file(GLOB COMPONENT_SOURCES "${SVO}/${component}/src/*.cpp")
  list(APPEND CORE_SOURCES ${COMPONENT_SOURCES})
endforeach()
list(APPEND CORE_SOURCES "${SVO}/svo_img_align/src/sparse_img_align_base.cpp" "${SVO}/svo_img_align/src/sparse_img_align.cpp")
foreach(component vikit_common vikit_cameras)
  file(GLOB COMPONENT_SOURCES "${SVO}/vikit/${component}/src/*.cpp")
  list(APPEND CORE_SOURCES ${COMPONENT_SOURCES})
endforeach()
add_library(schurvins_core STATIC ${CORE_SOURCES})
target_link_libraries(schurvins_core PUBLIC ${OpenCV_LIBRARIES} ${YAML_CPP_LIBRARY} glog ${Boost_LIBRARIES} Threads::Threads "${SCHUR_DEP_LIB}/libopengv.so" "${SCHUR_DEP_LIB}/librandom_generators.so")
add_executable(schurvins_headless main.cpp factory.cpp)
target_link_libraries(schurvins_headless PRIVATE schurvins_core)
install(TARGETS schurvins_headless RUNTIME DESTINATION bin)
